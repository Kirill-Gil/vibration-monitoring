    async function setMode(mode, btn) {
        document.querySelectorAll('.control-panel .btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        await fetch('/api/simulation/mode', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({ mode: mode })
        });
    }

    async function updateDashboard() {
        try {
            const res = await fetch('/api/telemetry');
            const data = await res.json();

            // Обновление общего статуса
            const statusElem = document.getElementById('general-status');
            statusElem.className = 'status-pill ' + data.status;
            if (data.status === 'NORMAL') statusElem.innerText = `ЗОНА A/B: НОРМА (${data.max_rms} мм/с)`;
            else if (data.status === 'WARNING') statusElem.innerText = `ЗОНА C: ПРЕДУПРЕЖДЕНИЕ (${data.max_rms} мм/с)`;
            else statusElem.innerText = `ЗОНА D: АВАРИЯ / ОСТАНОВ (${data.max_rms} мм/с)`;

            // Обновление 12 каналов мнемосхемы
            for (const [key, val] of Object.entries(data.rms_channels)) {
                const el = document.getElementById('val-' + key);
                if (el) el.innerText = val.toFixed(1);
            }

            // Подсветка опор
            ['ED_DE', 'ED_NDE', 'PUMP_DE', 'PUMP_NDE'].forEach(sup => {
                const node = document.getElementById('node-' + sup);
                const supMax = Math.max(
                    data.rms_channels[`${sup}_H`],
                    data.rms_channels[`${sup}_V`],
                    data.rms_channels[`${sup}_A`]
                );
                node.classList.remove('warning', 'alarm');
                if (supMax > 4.5) node.classList.add('alarm');
                else if (supMax > 2.8) node.classList.add('warning');
            });

            // Обновление графика огибающей
            Plotly.update('envelope-chart', {
                x: [data.envelope.freqs],
                y: [data.envelope.amps]
            });
            document.getElementById('bpfo-peak-label').innerText = 
                `Пик: ${data.envelope.peak_freq} Гц (Расчет BPFO: ${data.envelope.expected_bpfo} Гц)`;

            // Обновление MCSA
            Plotly.update('mcsa-chart', {
                x: [data.mcsa.freqs],
                y: [data.mcsa.db]
            });
            document.getElementById('currents-label').innerText = 
                `Фазы: L1=${data.currents.L1}A, L2=${data.currents.L2}A, L3=${data.currents.L3}A`;

            // Обновление алертов
            if (data.alerts && data.alerts.length > 0) {
                let html = '';
                data.alerts.forEach(a => {
                    const color = a.level === 'ALARM' ? 'var(--status-red)' : 'var(--status-yellow)';
                    html += `<tr>
                        <td>${a.time}</td>
                        <td style="color:${color}; font-weight:700;">${a.level}</td>
                        <td>${a.max_rms.toFixed(1)}</td>
                        <td>${a.desc}</td>
                    </tr>`;
                });
                document.getElementById('alert-tbody').innerHTML = html;
            }

        } catch (e) {
            console.error('Ошибка обновления телеметрии:', e);
        }
    }

    async function uploadFile(input) {
        if (!input.files || input.files.length === 0) return;
        const formData = new FormData();
        formData.append('file', input.files[0]);

        const resBox = document.getElementById('file-result');
        resBox.style.display = 'block';
        resBox.innerHTML = '<i>Выполняется преобразование Гильберта и спектральный анализ...</i>';

        try {
            const resp = await fetch('/api/diagnose/file', {
                method: 'POST',
                body: formData
            });
            const res = await resp.json();
            if (res.status === 'success') {
                resBox.innerHTML = `
                    <div style="background: #ECFDF5; border: 1px solid #10B981; padding: 12px; border-radius: 6px;">
                        <b>Результат верификации (${res.filename}):</b><br>
                        • Отсчетов: ${res.samples_count} (Fs = ${res.sampling_rate} Гц)<br>
                        • Расчетная частота дефекта (BPFO): <b>${res.calculated_bpfo} Гц</b><br>
                        • Локализованный пик алгоритма: <b style="color: #0078D2;">${res.detected_peak} Гц</b><br>
                        • Абсолютная погрешность: <b>${res.deviation_hz} Гц</b><br>
                        <i>${res.conclusion}</i>
                    </div>
                `;
            } else {
                resBox.innerHTML = `<span style="color:red">Ошибка: ${res.detail}</span>`;
            }
        } catch (err) {
            resBox.innerHTML = `<span style="color:red">Сбой загрузки: ${err.message}</span>`;
        }
    }

    // Запуск цикла опроса телеметрии раз в секунду
    setInterval(updateDashboard, 1000);
    updateDashboard();
